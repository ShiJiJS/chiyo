cmake_minimum_required(VERSION 3.20)
project(chiyo_ssh_session LANGUAGES CXX)
add_executable(ssh_session WIN32 main.cpp)
target_compile_features(ssh_session PRIVATE cxx_std_17)
target_compile_definitions(ssh_session PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(ssh_session PRIVATE /W4 /WX /EHsc /utf-8)
target_link_libraries(ssh_session PRIVATE advapi32 bcrypt shell32 windowsapp)
set_target_properties(ssh_session PROPERTIES OUTPUT_NAME info-ssh-session)
set_property(TARGET ssh_session PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(CHIYO_SSH_BUILD_TESTS "Build synthetic process-identity fixture (never bundled)" OFF)
if(CHIYO_SSH_BUILD_TESTS)
  add_executable(ssh_session_test_child WIN32 test_child.cpp)
  target_compile_features(ssh_session_test_child PRIVATE cxx_std_17)
  target_compile_definitions(ssh_session_test_child PRIVATE UNICODE _UNICODE NOMINMAX)
  target_compile_options(ssh_session_test_child PRIVATE /W4 /WX /EHsc /utf-8)
  target_link_libraries(ssh_session_test_child PRIVATE shell32)
endif()
