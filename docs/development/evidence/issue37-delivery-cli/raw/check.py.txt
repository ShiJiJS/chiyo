from pathlib import Path
import json,os,subprocess,sys
sys.stdout.reconfigure(encoding='utf-8')
root=Path.cwd();assert (root/'apps/desktop/pubspec.yaml').is_file()
out=root/'build/qa-issue37-delivery-cli'/sys.argv[1];out.mkdir(parents=True,exist_ok=False)
dart='C:/Users/Todoki/.local/share/flutter/bin/dart.bat'
env=os.environ.copy();env['PUB_HOSTED_URL']='https://pub.flutter-io.cn';env['FLUTTER_STORAGE_BASE_URL']='https://storage.flutter-io.cn'
packages=['packages/info_core','packages/info_storage','packages/info_cli','apps/desktop']
locks={root/p/'pubspec.lock':(root/p/'pubspec.lock').read_bytes() for p in packages}
phases=[]
def run(package,name,args):
    assert (root/package/'pubspec.yaml').is_file()
    with (out/(name+'.txt')).open('wb') as log:
        result=subprocess.run([dart,*args],cwd=root/package,env=env,stdout=log,stderr=subprocess.STDOUT)
    phases.append({'phase':name,'exitCode':result.returncode});print(json.dumps(phases[-1]),flush=True)
    if result.returncode:
        print((out/(name+'.txt')).read_text(encoding='utf-8',errors='replace')[-10000:],flush=True)
        raise RuntimeError(name)
try:
    for p,b in locks.items():p.write_bytes(b.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    for package,name in [('packages/info_core','core'),('packages/info_cli','cli')]:
        if '--core-only' in sys.argv and name!='core':continue
        run(package,name+'-pub',['pub','get','--enforce-lockfile','--offline'])
        changed=subprocess.check_output(['git','diff','--name-only','--',package+'/lib',package+'/test'],cwd=root,text=True).splitlines()
        changed+=subprocess.check_output(['git','ls-files','--others','--exclude-standard','--',package+'/lib',package+'/test'],cwd=root,text=True).splitlines()
        files=[str(Path(p).relative_to(package)) for p in changed if p.endswith('.dart')]
        if files:run(package,name+'-format',['format',*files])
        run(package,name+'-analyze',['analyze','--fatal-infos'])
        if '--test' in sys.argv:
            args=['test','--reporter','expanded']
            if '--name' in sys.argv:args+=['test/information_delivery_test.dart','--name',sys.argv[sys.argv.index('--name')+1]]
            run(package,name+'-tests',args)
finally:
    unexpected=[]
    norm=lambda b:b.replace(b'https://pub.flutter-io.cn',b'https://pub.dev').replace(b'\r\n',b'\n')
    for p,b in locks.items():
        if norm(p.read_bytes())!=norm(b):unexpected.append(str(p))
        else:p.write_bytes(b)
    (out/'result.json').write_text(json.dumps({'phases':phases,'unexpectedLocks':unexpected},indent=2)+'\n',encoding='utf-8',newline='\n')
    assert not unexpected,unexpected
