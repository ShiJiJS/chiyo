// Explicit format-compatibility fixtures, not genuine archived old writers.
// Recovery/database keys stay in this process and private synthetic files.
import 'dart:convert';
import 'dart:io';
import 'package:info_core/info_core.dart';
import 'package:info_storage/info_storage.dart';
import 'package:sqlite3/sqlite3.dart' as sqlite;

Future<void> main(List<String> args) async {
  if (args.length != 3 || !{'prepare', 'inspect'}.contains(args.first)) {
    throw StateError('Explicit compatibility fixture action required');
  }
  final root = Directory(args[1]);
  final canonical = await root.resolveSymbolicLinks();
  final temp = await Directory.systemTemp.resolveSymbolicLinks();
  if (root.absolute.path.toLowerCase() != canonical.toLowerCase() ||
      !canonical.toLowerCase().startsWith(
        '$temp${Platform.pathSeparator}'.toLowerCase(),
      ) ||
      !root.uri.pathSegments
          .where((s) => s.isNotEmpty)
          .last
          .startsWith('aim-packaged-compat-')) {
    throw StateError('Not an owned synthetic compatibility fixture');
  }
  final selected = int.parse(args[2]);
  if (![1, 2, 3, 4, 8].contains(selected)) {
    throw StateError('Unsupported fixture format');
  }
  final expectedFile = File('${root.path}/expected.json');
  Future<EncryptedInformationRepository> open(String profile) {
    final dir = '${root.path}/$profile/AgentInfoManage';
    return EncryptedInformationRepository.open(
      dir,
      WindowsDatabaseKeyStore('$dir/database-key.dpapi'),
    );
  }

  Future<void> export(EncryptedInformationRepository repo, String name) async {
    await repo.backups.enable();
    await repo.backups.saveRecoveryKeyTo('${root.path}/$name.recovery.txt');
    await repo.backups.exportTo('${root.path}/$name.aimbackup');
  }

  if (args.first == 'prepare') {
    if (!await root.list().isEmpty) throw StateError('Fixture is not new');
    final repo = await open('source');
    final service = InformationService(repository: repo);
    try {
      await service.owner.setNotificationsEnabled(false);
      await service.owner.setRequestLifetime(const Duration(minutes: 23));
      InformationItem item;
      final expected = <String, Object?>{'format': selected};
      if (selected == 8) {
        await service.owner.createTemplate('空文件模板', '', const [
          TemplateFieldInput('文件', '', kind: FieldKind.file),
        ]);
        InformationAttachmentRef? old;
        InformationItem? before;
        for (var version = 1; version <= 2; version++) {
          final file = File('${root.path}/synthetic-v$version.txt');
          await file.writeAsString('格式8虚构附件第$version版，保留原始字节。');
          final draft = await service.owner.beginInformationFileDraft();
          try {
            final stage = await service.owner.stageInformationFile(
              draft,
              file.path,
            );
            before = await service.owner.commitInformationDraft(
              draft,
              'compat-v$version',
              InformationDraftInput(
                name: '格式8虚构文件',
                description: '格式兼容验收，不是真实个人资料',
                itemId: before?.id,
                baseVersion: before?.version,
                fields: [
                  InformationDraftField.stagedFile(
                    '文件',
                    '',
                    stage.id,
                    fieldId: before?.fields.single.id,
                  ),
                ],
              ),
            );
            if (version == 1) old = before.fields.single.attachment;
          } finally {
            await service.owner.releaseInformationFileDraft(draft);
          }
        }
        item = before!;
        expected['oldAttachment'] = old!.toJson();
        expected['currentAttachment'] = item.fields.single.attachment!.toJson();
      } else {
        item = await service.owner.createInformation('格式兼容虚构资料', '', [
          const FieldInput('值', '', 'synthetic-before'),
        ]);
        await service.owner.correctValue(
          FieldRef(item.id, item.fields.single.id),
          'synthetic-current',
        );
        item = service.owner.information.single;
      }
      expected['itemId'] = item.id;
      await export(repo, 'selected');
      if (selected != 8) {
        final key = (await File(
          '${root.path}/selected.recovery.txt',
        ).readAsString()).trim();
        final decoded = base64Url.decode(
          key.substring('aim-recovery-v1:'.length),
        );
        final db = sqlite.sqlite3.open('${root.path}/selected.aimbackup');
        try {
          final hex = decoded
              .map((n) => n.toRadixString(16).padLeft(2, '0'))
              .join();
          db.execute('PRAGMA key = "x\'$hex\'"');
          final payload =
              jsonDecode(
                    db
                            .select('SELECT payload FROM backup_snapshot')
                            .single['payload']
                        as String,
                  )
                  as Map<String, dynamic>;
          payload['version'] = selected;
          final tables = payload['tables'] as Map;
          for (final name in [
            'managed_credentials',
            'credential_materials',
            'ssh_host_trust',
            'credential_use_requests',
            'credential_executions',
            'credential_focused_fills',
          ]) {
            tables.remove(name);
          }
          if (selected < 4) tables.remove('request_history');
          if (selected < 3) {
            for (final row in tables['information_items'] as List) {
              (row as Map).remove('created_at');
              row.remove('modified_at');
            }
            for (final row in tables['information_revisions'] as List) {
              final old = jsonDecode(row['payload'] as String) as Map;
              old.remove('createdAt');
              old.remove('modifiedAt');
              row['payload'] = jsonEncode(old);
            }
          }
          if (selected == 1) {
            for (final name in [
              'information_revisions',
              'recycle_entries',
              'recycle_facts',
              'revision_purge_facts',
              'grant_history',
              'access_records',
            ]) {
              tables.remove(name);
            }
          }
          db.execute('UPDATE backup_snapshot SET payload=?', [
            jsonEncode(payload),
          ]);
        } finally {
          db.close();
          decoded.fillRange(0, decoded.length, 0);
        }
      }
      await expectedFile.writeAsString(jsonEncode(expected));
      final key = (await File(
        '${root.path}/selected.recovery.txt',
      ).readAsString()).trim();
      final preview = await repo.backups.inspect(
        '${root.path}/selected.aimbackup',
        key,
      );
      if (preview.formatVersion != selected || preview.informationCount != 1) {
        throw StateError('Fixture format mismatch');
      }
      stdout.writeln(
        jsonEncode({
          'status': 'prepared',
          'format': preview.formatVersion,
          'information': preview.informationCount,
          'revisions': preview.revisionCount,
          'attachments': preview.attachmentCount,
          'attachmentBytes': preview.attachmentBytes,
          'templates': preview.templateCount,
          'credentials': preview.credentialCount,
        }),
      );
    } finally {
      await service.close();
      await repo.close();
    }
    // Nondefault empty target settings must not prevent restore.
    final target = await open('target');
    final targetService = InformationService(repository: target);
    try {
      await targetService.owner.setNotificationsEnabled(false);
      await targetService.owner.setRequestLifetime(const Duration(minutes: 17));
    } finally {
      await targetService.close();
      await target.close();
    }
    return;
  }
  final expected = jsonDecode(await expectedFile.readAsString()) as Map;
  final repo = await open('target');
  final service = InformationService(repository: repo);
  try {
    final item = repo.items.single;
    var originalsMatch =
        selected == 8 || item.fields.single.value == 'synthetic-current';
    if (selected == 8) {
      originalsMatch = true;
      for (final (version, label) in [
        (1, 'oldAttachment'),
        (2, 'currentAttachment'),
      ]) {
        final ref = InformationAttachmentRef.fromJson(expected[label]);
        final before = await File(
          '${root.path}/synthetic-v$version.txt',
        ).readAsBytes();
        final after = await repo.readInformationAttachmentChunk(
          ref,
          0,
          before.length,
        );
        originalsMatch =
            originalsMatch &&
            before.length == after.length &&
            List.generate(
              before.length,
              (i) => before[i] == after[i],
            ).every((same) => same);
      }
    }
    stdout.writeln(
      jsonEncode({
        'status': 'inspected',
        'sameIdentity': item.id == expected['itemId'],
        'version': item.version,
        'originalsMatch': originalsMatch,
        'datesUnknown': item.createdAt == null && item.modifiedAt == null,
        'revisions': repo.informationRevisions.length,
        'templates': service.owner.templates.length,
        'credentials': repo.managedCredentials.length,
        'requests': service.owner.requests.length,
        'activeGrants': repo.grantHistory.where((g) => g.active).length,
        'fileDeliveries': repo.informationFileDeliveries.length,
        'requestLifetimeMinutes': service.owner.requestLifetime.inMinutes,
      }),
    );
  } finally {
    await service.close();
    await repo.close();
  }
}
