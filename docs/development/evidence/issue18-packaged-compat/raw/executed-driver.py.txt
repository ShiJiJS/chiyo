"""Operator-assisted native restore of explicit format fixtures into empty profiles.

Formats1-4 are compatibility fixtures derived from the current authenticated
snapshot using the same projections as the repository tests. They are NOT backups
produced by genuine old releases. Format8 contains two ordinary file versions and
an empty file template, with no file delivery facts. All data is synthetic.

Requires port47125 free. In each printed phase, select the indicated backup and
recovery file in the actual app's native Open dialogs, verify its format/counts,
confirm restore into the empty test library, and only then write the observation
receipt with {"observed": true}. Never read/print/attach recovery keys. The receipt
records a real operator observation, not automatic image recognition.

This exits only its own child through the packaged CLI and explicit PID. A failed
run preserves its private synthetic root for diagnosis; there is no force-kill.
"""
from pathlib import Path
import argparse
import hashlib
import json
import os
import queue
import shutil
import socket
import subprocess
import sys
import tempfile
import threading
import time

sys.stdout.reconfigure(encoding="utf-8")
parser = argparse.ArgumentParser(description=__doc__)
for name in ["delivery", "dart", "evidence"]:
    parser.add_argument("--" + name, type=Path, required=True)
parser.add_argument("--formats", type=int, nargs="+", default=[1, 2, 3, 4, 8])
args = parser.parse_args()
if sys.platform != "win32" or any(n not in [1, 2, 3, 4, 8] for n in args.formats):
    parser.error("Windows and explicit supported formats are required")
root = Path(__file__).resolve().parent.parent
out = args.evidence.resolve()
out.mkdir(parents=True, exist_ok=False)
payload = args.delivery.resolve(strict=True)
manifest = json.loads((payload / "manifest.json").read_text(encoding="utf-8-sig"))
for entry in manifest["files"]:
    file = (payload / entry["path"]).resolve(strict=True)
    assert file.is_relative_to(payload)
    assert file.stat().st_size == entry["bytes"]
    assert hashlib.sha256(file.read_bytes()).hexdigest() == entry["sha256"]
app, cli = payload / "app/info_desktop.exe", payload / "cli/info.exe"
dart = str(args.dart.resolve(strict=True))
environment = os.environ.copy()
environment.update(PUB_HOSTED_URL="https://pub.flutter-io.cn", FLUTTER_STORAGE_BASE_URL="https://storage.flutter-io.cn")
locks = {root / name / "pubspec.lock": (root / name / "pubspec.lock").read_bytes()
         for name in ["packages/info_core", "packages/info_cli", "packages/info_storage", "apps/desktop"]}
normalize = lambda b: b.replace(b"https://pub.flutter-io.cn", b"https://pub.dev").replace(b"\r\n", b"\n")
facts = {"productSource": manifest["sourceCommit"], "complete": False, "checks": [], "formats": []}
streams, temporary = [], []
process = agent = None


def check(name, condition, **data):
    facts["checks"].append(dict(check=name, passed=bool(condition), **data))
    assert condition, name


def phase(stage, **data):
    value = dict(phase=stage, **data)
    text = json.dumps(value, ensure_ascii=False)
    (out / "phase.json").write_text(text + "\n", encoding="utf-8")
    with (out / "phases.jsonl").open("a", encoding="utf-8") as log:
        log.write(text + "\n")
    print(text, flush=True)


def run(name, arguments):
    with (out / (name + ".txt")).open("wb") as log:
        result = subprocess.run([dart, *arguments], cwd=root / "packages/info_storage", env=environment,
                                stdout=log, stderr=subprocess.STDOUT, timeout=150)
    check(name, result.returncode == 0)
    text = (out / (name + ".txt")).read_text(encoding="utf-8", errors="replace")
    start = text.find('{"status":')
    return json.loads(text[start:].strip()) if start >= 0 else None


class Agent:
    def __init__(self, number):
        log = (out / f"agent-{number}-stderr.txt").open("wb")
        streams.append(log)
        self.child = subprocess.Popen([str(cli)], cwd=cli.parent, stdin=subprocess.PIPE,
                                      stdout=subprocess.PIPE, stderr=log, text=True, encoding="utf-8",
                                      creationflags=subprocess.CREATE_NO_WINDOW)
        self.lines = queue.Queue()

        def read():
            for line in self.child.stdout:
                self.lines.put(line)
        threading.Thread(target=read, daemon=True).start()

    def call(self, command, **data):
        self.child.stdin.write(json.dumps(dict(command=command, **data)) + "\n")
        self.child.stdin.flush()
        return json.loads(self.lines.get(timeout=45))

    def close(self):
        self.child.stdin.close()
        assert self.child.wait(timeout=10) == 0


def stop():
    result = subprocess.run([str(cli), "app", "quit", "--pid", str(process.pid)], capture_output=True,
                            text=True, encoding="utf-8", timeout=65, creationflags=subprocess.CREATE_NO_WINDOW)
    value = json.loads(result.stdout)
    check("own-child-normal-exit", result.returncode == 0 and value["status"] == "exited"
          and process.wait(timeout=5) == 0, processId=process.pid)


try:
    for file, before in locks.items():
        file.write_bytes(before.replace(b"https://pub.dev", b"https://pub.flutter-io.cn"))
    run("fixture-format", ["format", "test/support/packaged_compat_fixture.dart"])
    run("pub", ["pub", "get", "--offline", "--enforce-lockfile"])
    run("analyze", ["analyze", "--fatal-infos"])
    for number in args.formats:
        temp = Path(tempfile.mkdtemp(prefix="aim-packaged-compat-")).resolve()
        temporary.append(temp)
        fixture = run(f"prepare-{number}", ["run", "test/support/packaged_compat_fixture.dart", "prepare", str(temp), str(number)])
        check("fixture-format-exact", fixture["format"] == number, format=number)
        backup, recovery = temp / "selected.aimbackup", temp / "selected.recovery.txt"
        before = {file: hashlib.sha256(file.read_bytes()).hexdigest() for file in [backup, recovery]}
        with socket.socket() as available:
            available.bind(("127.0.0.1", 47125))
        child_env = environment.copy()
        child_env["LOCALAPPDATA"] = str(temp / "target")
        log = (out / f"app-{number}.txt").open("wb")
        streams.append(log)
        process = subprocess.Popen([str(app)], cwd=app.parent, env=child_env,
                                   stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT)
        agent = Agent(number)
        end = time.monotonic() + 30
        while time.monotonic() < end:
            if agent.call("capabilities")["status"] == "ok":
                break
            assert process.poll() is None
            time.sleep(.2)
        else:
            raise RuntimeError("Packaged service did not become ready")
        check("new-empty-session", agent.call("session", chatLabel="Synthetic format restore")["status"] == "ok")
        check("empty-before-restore", not agent.call("catalog")["items"])
        receipt = out / f"observed-format-{number}.json"
        phase("restore-format-via-native-dialogs", format=number, pid=process.pid, backup=str(backup),
              recovery=str(recovery), expected=fixture, receipt=str(receipt))
        deadline = time.monotonic() + 600
        while not receipt.is_file():
            assert time.monotonic() < deadline, "Bounded real operator observation missing"
            assert process.poll() is None
            time.sleep(.2)
        check("actual-UI-observed", json.loads(receipt.read_text(encoding="utf-8"))["observed"], format=number)
        check("service-still-ready", agent.call("capabilities")["status"] == "ok")
        denied = agent.call("session-check")
        check("pre-restore-session-invalidated", denied["status"] == "unauthorized", result=denied, format=number)
        stop()
        agent.close()
        agent = None
        inspected = run(f"inspect-{number}", ["run", "test/support/packaged_compat_fixture.dart", "inspect", str(temp), str(number)])
        check("restored-originals-and-facts", inspected["sameIdentity"] and inspected["version"] == 2
              and inspected["originalsMatch"] and inspected["credentials"] == 0 and inspected["requests"] == 0
              and inspected["activeGrants"] == 0 and inspected["fileDeliveries"] == 0
              and inspected["requestLifetimeMinutes"] == 23
              and inspected["revisions"] == (0 if number == 1 else 1)
              and inspected["templates"] == (1 if number == 8 else 0), format=number)
        check("dates-have-correct-legacy-meaning", inspected["datesUnknown"] == (number in [1, 2]), format=number)
        check("source-backup-and-key-unchanged", all(hashlib.sha256(file.read_bytes()).hexdigest() == digest
              for file, digest in before.items()), format=number)
        facts["formats"].append(dict(format=number, fixture=fixture, inspection=inspected, complete=True))
    facts["complete"] = True
    phase("complete")
finally:
    if process is not None and process.poll() is None:
        stop()
    if agent is not None:
        agent.close()
    for log in streams:
        log.close()
    bad = []
    for file, before in locks.items():
        if normalize(file.read_bytes()) != normalize(before):
            bad.append(str(file))
        else:
            file.write_bytes(before)
    facts["unexpectedLocks"] = bad
    if facts["complete"]:
        for temp in temporary:
            assert temp.parent == Path(tempfile.gettempdir()).resolve() and temp.name.startswith("aim-packaged-compat-")
            assert not temp.is_symlink() and not temp.is_junction()
            assert not any(file.is_symlink() or file.is_junction() for file in temp.rglob("*"))
            shutil.rmtree(temp)
        facts["temporaryRemoved"] = all(not temp.exists() for temp in temporary)
    else:
        facts["preservedTemporary"] = [str(temp) for temp in temporary]
    (out / "result.json").write_text(json.dumps(facts, indent=2) + "\n", encoding="utf-8")
    assert not bad, bad
