// Synthetic crash acceptance only. Never emit database/recovery keys or bodies.
import 'dart:convert';
import 'dart:io';
import 'dart:math';
import 'dart:typed_data';
import 'package:info_core/info_core.dart';
import 'package:info_storage/info_storage.dart';
import 'package:sqlite3/sqlite3.dart' as sqlite;

const fileCount = 10;
const fileBytes = 8 * 1024 * 1024;

Future<void> main(List<String> args) async {
  if (args.length != 2 ||
      !{
        'prepare',
        'inspect-rollback',
        'inspect-restored',
      }.contains(args.first)) {
    throw StateError('Explicit synthetic restore-crash action required');
  }
  final root = Directory(args[1]);
  final actual = await root.resolveSymbolicLinks();
  final temp = await Directory.systemTemp.resolveSymbolicLinks();
  if (root.absolute.path.toLowerCase() != actual.toLowerCase() ||
      !actual.toLowerCase().startsWith(
        '$temp${Platform.pathSeparator}'.toLowerCase(),
      ) ||
      !root.uri.pathSegments
          .where((s) => s.isNotEmpty)
          .last
          .startsWith('aim-packaged-restore-crash-')) {
    throw StateError('Not an owned synthetic restore-crash fixture');
  }
  Future<EncryptedInformationRepository> open(String name) {
    final directory = '${root.path}/$name/AgentInfoManage';
    return EncryptedInformationRepository.open(
      directory,
      WindowsDatabaseKeyStore('$directory/database-key.dpapi'),
    );
  }

  if (args.first == 'prepare') {
    if (!await root.list().isEmpty) throw StateError('Fixture is not new');
    final repo = await open('source');
    final service = InformationService(repository: repo);
    final expected = <Map<String, Object?>>[];
    try {
      await service.owner.setNotificationsEnabled(false);
      await service.owner.setRequestLifetime(const Duration(minutes: 23));
      for (var index = 1; index <= fileCount; index++) {
        final file = File('${root.path}/synthetic-$index.bin');
        final bytes = Uint8List(fileBytes)..fillRange(0, fileBytes, index);
        await file.writeAsBytes(bytes);
        final draft = await service.owner.beginInformationFileDraft();
        try {
          final staged = await service.owner.stageInformationFile(
            draft,
            file.path,
          );
          final item = await service.owner.commitInformationDraft(
            draft,
            'restore-crash-$index',
            InformationDraftInput(
              name: 'Synthetic attachment $index',
              description: 'Disposable restore interruption acceptance',
              fields: [InformationDraftField.stagedFile('File', '', staged.id)],
            ),
          );
          expected.add({
            'itemId': item.id,
            'fieldId': item.fields.single.id,
            'attachment': item.fields.single.attachment!.toJson(),
            'pattern': index,
          });
        } finally {
          await service.owner.releaseInformationFileDraft(draft);
        }
      }
      await repo.backups.enable();
      await repo.backups.saveRecoveryKeyTo(
        '${root.path}/synthetic.recovery.txt',
      );
      final summary = await repo.backups.exportTo(
        '${root.path}/synthetic.aimbackup',
      );
      if (summary.attachmentCount != fileCount ||
          summary.attachmentBytes != fileCount * fileBytes) {
        throw StateError('Synthetic attachment fixture count mismatch');
      }
      await File(
        '${root.path}/expected.json',
      ).writeAsString(jsonEncode(expected));
    } finally {
      await service.close();
      await repo.close();
    }
    final target = await open('target');
    final targetService = InformationService(repository: target);
    try {
      await targetService.owner.setNotificationsEnabled(false);
      await targetService.owner.setRequestLifetime(const Duration(minutes: 17));
    } finally {
      await targetService.close();
      await target.close();
    }
    stdout.writeln(
      jsonEncode({
        'status': 'prepared',
        'format': 8,
        'files': fileCount,
        'bytes': fileCount * fileBytes,
      }),
    );
    return;
  }

  final directory = '${root.path}/target/AgentInfoManage';
  final key = await WindowsDatabaseKeyStore(
    '$directory/database-key.dpapi',
  ).read();
  if (key == null) throw StateError('Missing synthetic target key');
  final raw = sqlite.sqlite3.open('$directory/library.db');
  late Map<String, Object?> rawFacts;
  try {
    final hex = key.map((n) => n.toRadixString(16).padLeft(2, '0')).join();
    raw.execute('PRAGMA key = "x\'$hex\'"');
    // The first read performs any required hot-journal recovery before SDK open.
    rawFacts = {
      'journalMode': raw.select('PRAGMA journal_mode').single.values.single,
      'integrity': raw.select('PRAGMA integrity_check').single.values.single,
      'information': raw
          .select('SELECT count(*) AS n FROM information_items')
          .single['n'],
      'attachmentVersions': raw
          .select('SELECT count(*) AS n FROM information_attachment_versions')
          .single['n'],
      'attachmentBytes': raw
          .select(
            'SELECT coalesce(sum(byte_length),0) AS n FROM information_attachment_versions',
          )
          .single['n'],
      'fileFields': raw
          .select('SELECT count(*) AS n FROM information_file_fields')
          .single['n'],
      'stagedFiles': raw
          .select('SELECT count(*) AS n FROM information_file_stages')
          .single['n'],
    };
  } finally {
    raw.close();
    key.fillRange(0, key.length, 0);
  }
  final repo = await open('target');
  final service = InformationService(repository: repo);
  try {
    var originalsMatch = true;
    if (args.first == 'inspect-restored') {
      final expected =
          jsonDecode(await File('${root.path}/expected.json').readAsString())
              as List;
      originalsMatch = repo.items.length == expected.length;
      for (final value in expected) {
        final row = value as Map<String, dynamic>;
        final item = repo.items.singleWhere((item) => item.id == row['itemId']);
        final ref = InformationAttachmentRef.fromJson(
          row['attachment'] as Map<String, dynamic>,
        );
        originalsMatch =
            originalsMatch &&
            item.fields.single.id == row['fieldId'] &&
            item.fields.single.attachment == ref;
        for (var offset = 0; offset < fileBytes; offset += 1024 * 1024) {
          final count = min(1024 * 1024, fileBytes - offset);
          final chunk = await repo.readInformationAttachmentChunk(
            ref,
            offset,
            count,
          );
          originalsMatch =
              originalsMatch &&
              chunk.length == count &&
              chunk.every((byte) => byte == row['pattern']);
        }
      }
    }
    stdout.writeln(
      jsonEncode({
        'status': 'inspected',
        'action': args.first,
        ...rawFacts,
        'modelInformation': repo.items.length,
        'requestLifetimeMinutes': service.owner.requestLifetime.inMinutes,
        'requests': service.owner.requests.length,
        'activeGrants': repo.grantHistory.where((grant) => grant.active).length,
        'fileDeliveries': repo.informationFileDeliveries.length,
        'originalsMatch': originalsMatch,
      }),
    );
  } finally {
    await service.close();
    await repo.close();
  }
}
