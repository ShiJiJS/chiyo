late IOSink trace;
final elapsed = Stopwatch()..start();
String phase = 'startup';
int batch = 0;
final nodes = <int, Map<String, Object?>>{};
void record(String type, Map<String, Object?> data) {
  final row = {
    'utc': DateTime.now().toUtc().toIso8601String(),
    'ms': elapsed.elapsedMilliseconds,
    'type': type,
    'phase': phase,
    ...data,
  };
  trace.writeln(jsonEncode(row));
}

void recordBatch(List<Map<String, Object?>> rows) {
  for (final row in rows) {
    nodes[row['id']! as int] = row;
  }
  final reached = <int>{};
  void visit(int id) {
    if (!reached.add(id)) return;
    for (final child in (nodes[id]?['children'] as List<int>?) ?? <int>[]) {
      visit(child);
    }
  }

  visit(0);
  final orphan = rows.where((r) => !reached.contains(r['id'])).toList();
  final number = ++batch;
  record('semantics-batch', {
    'batch': number,
    'updates': rows,
    'orphans': orphan,
    'orphanChildNodes': [
      for (final r in orphan)
        for (final id in r['children']! as List<int>) nodes[id],
    ],
    'reachableCount': reached.length,
  });
  stdout.writeln(
    'AX_BATCH $number ms=${elapsed.elapsedMilliseconds} phase=$phase updates=${rows.length} orphans=${orphan.map((r) => r['id']).toList()}',
  );
}

class ProbeBinding extends WidgetsFlutterBinding {
  @override
  ui.SemanticsUpdateBuilder createSemanticsUpdateBuilder() =>
      ProbeBuilder(super.createSemanticsUpdateBuilder());
}

class ProbeBuilder implements ui.SemanticsUpdateBuilder {
  ProbeBuilder(this.delegate);
  final ui.SemanticsUpdateBuilder delegate;
  final rows = <Map<String, Object?>>[];
  @override
  ui.SemanticsUpdate build() {
    recordBatch(rows);
    return delegate.build();
  }

  @override
  void updateCustomAction({
    required int id,
    String? label,
    String? hint,
    int overrideId = -1,
  }) => delegate.updateCustomAction(
    id: id,
    label: label,
    hint: hint,
    overrideId: overrideId,
  );
  @override
  void updateNode({
    required int id,
    required ui.SemanticsFlags flags,
    required int actions,
    required int maxValueLength,
    required int currentValueLength,
    required int textSelectionBase,
    required int textSelectionExtent,
    required int platformViewId,
    required int scrollChildren,
    required int scrollIndex,
    required int traversalParent,
    required double scrollPosition,
    required double scrollExtentMax,
    required double scrollExtentMin,
    required ui.Rect rect,
    required String identifier,
    required String label,
    required List<ui.StringAttribute> labelAttributes,
    required String value,
    required List<ui.StringAttribute> valueAttributes,
    required String increasedValue,
    required List<ui.StringAttribute> increasedValueAttributes,
    required String decreasedValue,
    required List<ui.StringAttribute> decreasedValueAttributes,
    required String hint,
    required List<ui.StringAttribute> hintAttributes,
    required String tooltip,
    required ui.TextDirection? textDirection,
    required Float64List transform,
    required Float64List hitTestTransform,
    required Int32List childrenInTraversalOrder,
    required Int32List childrenInHitTestOrder,
    required Int32List additionalActions,
    int headingLevel = 0,
    String linkUrl = '',
    ui.SemanticsRole role = ui.SemanticsRole.none,
    required List<String>? controlsNodes,
    ui.SemanticsValidationResult validationResult =
        ui.SemanticsValidationResult.none,
    ui.SemanticsHitTestBehavior hitTestBehavior =
        ui.SemanticsHitTestBehavior.defer,
    required ui.SemanticsInputType inputType,
    required ui.Locale? locale,
    required String minValue,
    required String maxValue,
  }) {
    delegate.updateNode(
      id: id,
      flags: flags,
      actions: actions,
      maxValueLength: maxValueLength,
      currentValueLength: currentValueLength,
      textSelectionBase: textSelectionBase,
      textSelectionExtent: textSelectionExtent,
      platformViewId: platformViewId,
      scrollChildren: scrollChildren,
      scrollIndex: scrollIndex,
      traversalParent: traversalParent,
      scrollPosition: scrollPosition,
      scrollExtentMax: scrollExtentMax,
      scrollExtentMin: scrollExtentMin,
      rect: rect,
      identifier: identifier,
      label: label,
      labelAttributes: labelAttributes,
      value: value,
      valueAttributes: valueAttributes,
      increasedValue: increasedValue,
      increasedValueAttributes: increasedValueAttributes,
      decreasedValue: decreasedValue,
      decreasedValueAttributes: decreasedValueAttributes,
      hint: hint,
      hintAttributes: hintAttributes,
      tooltip: tooltip,
      textDirection: textDirection,
      transform: transform,
      hitTestTransform: hitTestTransform,
      childrenInTraversalOrder: childrenInTraversalOrder,
      childrenInHitTestOrder: childrenInHitTestOrder,
      additionalActions: additionalActions,
      headingLevel: headingLevel,
      linkUrl: linkUrl,
      role: role,
      controlsNodes: controlsNodes,
      validationResult: validationResult,
      hitTestBehavior: hitTestBehavior,
      inputType: inputType,
      locale: locale,
      minValue: minValue,
      maxValue: maxValue,
    );

    rows.add({
      'id': id,
      'label': label,
      'tooltip': tooltip,
      'hint': hint,
      'children': childrenInTraversalOrder.toList(),
      'hitChildren': childrenInHitTestOrder.toList(),
      'traversalParent': traversalParent,
      'role': role.name,
      'valueLength': value.length,
      'rect': [rect.left, rect.top, rect.right, rect.bottom],
    });
  }
}

Future<void> main(List<String> arguments) async {
  if (!const bool.fromEnvironment('D4_AX_PROBE') || arguments.length != 1)
    exit(64);
  final configuration =
      jsonDecode(await File(arguments.single).readAsString()) as Map;
  final root = Directory(configuration['root'] as String);
  if (!root.absolute.path.toLowerCase().contains('ax-probe') ||
      !root.uri.pathSegments
          .where((p) => p.isNotEmpty)
          .last
          .startsWith('synthetic-') ||
      await root.resolveSymbolicLinks() != root.absolute.path)
    throw StateError('Synthetic AX fixture required');
  trace = File('${root.path}/semantics.jsonl').openWrite();
  final binding = ProbeBinding();
  binding.addSemanticsEnabledListener(
    () => record('semantics-enabled', {'enabled': binding.semanticsEnabled}),
  );
  record('binding-created', {'semanticsEnabled': binding.semanticsEnabled});
  final platform = NativeDesktopPlatform();
  final bootstrap = DesktopBootstrap(
    port: configuration['port'] as int,
    platform: platform,
    openLibrary: () async {
      final repo = await EncryptedInformationRepository.open(
        '${root.path}/library',
        WindowsDatabaseKeyStore('${root.path}/library/database-key.dpapi'),
      );
      final service = InformationService(repository: repo);
      if (service.owner.information.isEmpty) {
        await service.owner.createInformation(
          'AX synthetic information',
          'Only synthetic QA content.',
          const [FieldInput('Address', '', 'ax.example.invalid')],
        );
      }
      await service.owner.setNotificationsEnabled(false);
      return DesktopLibrary(
        service: service,
        backups: repo.backups,
        close: repo.close,
      );
    },
  );
  await bootstrap.start();
  if (bootstrap.state != DesktopStartupState.ready) exit(65);
  runApp(
    app.InformationApp(
      service: bootstrap.library!.service,
      port: bootstrap.boundPort!,
      runtime: bootstrap.runtime!,
      backups: bootstrap.library!.backups,
    ),
  );
  await File('${root.path}/status.json').writeAsString(
    jsonEncode({'pid': pid, 'port': bootstrap.boundPort, 'ready': true}),
    flush: true,
  );
  record('ready', {'pid': pid});
  var polling = false;
  String? nonce;
  Timer.periodic(const Duration(milliseconds: 100), (timer) async {
    if (polling) return;
    polling = true;
    try {
      final commandFile = File('${root.path}/control.json');
      if (await commandFile.exists()) {
        final command = jsonDecode(await commandFile.readAsString()) as Map;
        if (command['nonce'] != nonce) {
          nonce = command['nonce'] as String;
          phase = command['phase'] as String? ?? command['action'] as String;
          record('control', {'action': command['action'], 'nonce': nonce});
          if (command['action'] == 'request') {
            final service = bootstrap.library!.service;
            final item = service.owner.information.single;
            final session = service.agent.openSession(
              'AX synthetic conversation',
            );
            final request = service.agent.request(
              session,
              'AX synthetic request',
              [FieldRef(item.id, item.fields.single.id)],
            );
            record('request-created', {'requestId': request.id});
            await Future<void>.delayed(const Duration(milliseconds: 100));
            service.owner.approve(request.id, [
              FieldRef(item.id, item.fields.single.id),
            ]);
            record('request-approved', {'requestId': request.id});
          } else if (command['action'] == 'show') {
            await platform.showWindow();
          } else if (command['action'] == 'quit') {
            timer.cancel();
            await trace.flush();
            await bootstrap.runtime!.quit();
          }
          await File('${root.path}/ack.json').writeAsString(
            jsonEncode({'nonce': nonce, 'phase': phase}),
            flush: true,
          );
        }
      }
      await trace.flush();
    } finally {
      polling = false;
    }
  });
}
