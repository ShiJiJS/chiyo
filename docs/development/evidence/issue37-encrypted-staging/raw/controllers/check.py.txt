from pathlib import Path
import json,os,subprocess,sys
root=Path.cwd();assert (root/'apps/desktop/pubspec.yaml').is_file()
sdk=Path('C:/Users/Todoki/.local/share/flutter/bin/dart.bat')
out=root/'build/qa-issue37-staging/close-green';out.mkdir(exist_ok=False)
env=os.environ.copy();env['PUB_HOSTED_URL']='https://pub.flutter-io.cn';env['FLUTTER_STORAGE_BASE_URL']='https://storage.flutter-io.cn'
locks={root/p/'pubspec.lock':(root/p/'pubspec.lock').read_bytes() for p in ['packages/info_core','packages/info_storage','packages/info_cli','apps/desktop']}
phases=[]
def run(package,name,args):
    cwd=root/package;assert (cwd/'pubspec.yaml').is_file()
    with (out/(name+'.txt')).open('wb') as log:
        p=subprocess.run([str(sdk),*args],cwd=cwd,env=env,stdout=log,stderr=subprocess.STDOUT)
    phases.append({'phase':name,'exitCode':p.returncode})
    print(json.dumps(phases[-1]),flush=True)
    if p.returncode:
        print((out/(name+'.txt')).read_text(encoding='utf-8',errors='replace')[-9000:],flush=True)
        raise RuntimeError(name)
try:
    for path,data in locks.items():path.write_bytes(data.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    run('packages/info_core','core-format',['format','lib/info_core.dart','lib/information.dart','lib/information_files.dart','lib/information_file_access.dart','test/information_file_staging_test.dart'])
    run('packages/info_storage','storage-format',['format','lib/info_storage.dart','lib/information_files_storage.dart','lib/backups.dart','test/information_file_staging_test.dart','test/attachment_legacy_boundary_test.dart','test/credential_reuse_storage_test.dart','test/focused_fill_storage_test.dart'])
    for p in (['core'] if '--core-only' in sys.argv else ['storage'] if '--storage-only' in sys.argv else ['core','storage']):
        package='packages/info_'+p
        run(package,p+'-pub',['pub','get','--enforce-lockfile','--offline'])
        run(package,p+'-analyze',['analyze','--fatal-infos'])
        run(package,p+'-tests',['test','--reporter','expanded','test/information_file_staging_test.dart'])
finally:
    unexpected=[]
    for path,data in locks.items():
        norm=lambda b:b.replace(b'https://pub.flutter-io.cn',b'https://pub.dev').replace(b'\r\n',b'\n')
        if norm(path.read_bytes())!=norm(data):unexpected.append(str(path))
        else:path.write_bytes(data)
    (out/'result.json').write_text(json.dumps({'phases':phases,'unexpectedLocks':unexpected},indent=2)+'\n',encoding='utf-8',newline='\n')
    assert not unexpected,unexpected
