from pathlib import Path
import json,os,subprocess
root=Path.cwd();assert (root/'apps/desktop/pubspec.yaml').is_file()
out=root/'build/qa-issue37-staging/desktop-close';out.mkdir(exist_ok=False)
flutter='C:/Users/Todoki/.local/share/flutter/bin/flutter.bat'
env=os.environ.copy();env['PUB_HOSTED_URL']='https://pub.flutter-io.cn';env['FLUTTER_STORAGE_BASE_URL']='https://storage.flutter-io.cn'
locks={root/p/'pubspec.lock':(root/p/'pubspec.lock').read_bytes() for p in ['packages/info_cli','packages/info_core','packages/info_storage','apps/desktop']}
results=[]
try:
    for p,b in locks.items():p.write_bytes(b.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    for name,args in [('pub',['pub','get','--enforce-lockfile','--offline']),('tests',['test','--no-pub','--reporter','expanded','test/credential_reuse_widget_test.dart'])]:
        with (out/(name+'.txt')).open('wb') as f:
            r=subprocess.run([flutter,*args],cwd=root/'apps/desktop',env=env,stdout=f,stderr=subprocess.STDOUT,timeout=90)
        results.append({'phase':name,'exitCode':r.returncode});print(json.dumps(results[-1]),flush=True)
        if r.returncode:
            print((out/(name+'.txt')).read_text(encoding='utf-8',errors='replace')[-8000:],flush=True)
            raise RuntimeError(name)
finally:
    unexpected=[]
    norm=lambda b:b.replace(b'https://pub.flutter-io.cn',b'https://pub.dev').replace(b'\r\n',b'\n')
    for p,b in locks.items():
        if norm(p.read_bytes())!=norm(b):unexpected.append(str(p))
        else:p.write_bytes(b)
    (out/'result.json').write_text(json.dumps({'phases':results,'unexpectedLocks':unexpected},indent=2)+'\n',encoding='utf-8',newline='\n')
    assert not unexpected,unexpected
