import Cocoa
import FlutterMacOS

/// NSTextView's inherited selectAll: can update only the accessibility field
/// editor. Send the selector through Flutter's existing text-input bridge so
/// EditableText receives it as well. Native sheets keep their responder chain.
final class NativeSelectAllMenu: NSObject, NSMenuItemValidation {
  private weak var window: NSWindow?
  private let selectAll = NSSelectorFromString("selectAll:")

  init(window: NSWindow) {
    self.window = window
  }

  func install(in menu: NSMenu) {
    for item in menu.items {
      if let submenu = item.submenu { install(in: submenu) }
      if item.action == selectAll {
        print("QA_INSTALL_SELECT_ALL")
        fflush(stdout)
        item.target = self
        item.action = #selector(selectAllFromMenu(_:))
      }
    }
  }

  private var flutterEditor: NSTextView? {
    guard let window, NSApp.keyWindow === window, window.attachedSheet == nil,
          let editor = window.firstResponder as? NSTextView,
          editor.window === window,
          Bundle(for: type(of: editor)) == Bundle(for: FlutterViewController.self)
    else { return nil }
    return editor
  }

  @objc private func selectAllFromMenu(_ sender: NSMenuItem) {
    print("QA_SELECT_ALL_INVOKED key=\(NSApp.keyWindow === window) sheet=\(window?.attachedSheet != nil) responder=\(String(describing: window?.firstResponder))")
    if let responder = window?.firstResponder as? NSTextView {
      print("QA_EDITOR bundle=\(Bundle(for: type(of: responder)).bundleIdentifier ?? "nil") flutter=\(Bundle(for: FlutterViewController.self).bundleIdentifier ?? "nil") window=\(responder.window === window)")
    }
    fflush(stdout)
    if let editor = flutterEditor {
      editor.doCommand(by: selectAll)
    } else {
      NSApp.sendAction(selectAll, to: nil, from: sender)
    }
  }

  func validateMenuItem(_ menuItem: NSMenuItem) -> Bool {
    if flutterEditor != nil { return true }
    // Validate the original native action, not our forwarding selector.
    let native = NSMenuItem(title: menuItem.title, action: selectAll,
                            keyEquivalent: menuItem.keyEquivalent)
    guard let target = NSApp.target(forAction: selectAll, to: nil, from: native)
    else { return false }
    if let validator = target as? NSMenuItemValidation {
      return validator.validateMenuItem(native)
    }
    if let validator = target as? NSUserInterfaceValidations {
      return validator.validateUserInterfaceItem(native)
    }
    return true
  }
}
