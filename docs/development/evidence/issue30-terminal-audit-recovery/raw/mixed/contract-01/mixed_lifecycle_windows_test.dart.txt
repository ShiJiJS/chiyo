import 'dart:convert';
import 'dart:io';
import 'package:info_core/info_core.dart';
import 'package:test/test.dart';
import 'support/mixed_lifecycle.dart';

void main() {
  late LifecycleAdapters adapters;
  setUpAll(() async {
    if (Platform.isWindows) adapters = await LifecycleAdapters.build();
  });
  tearDownAll(() async {
    if (Platform.isWindows) await adapters.close();
  });

  Future<MixedLifecycle> fixture() async {
    final f = MixedLifecycle(adapters);
    await f.start();
    await f.approveThree();
    return f;
  }

  Future<void> assertRedactedOutput(MixedLifecycle f) async {
    var text = '';
    await f.until(() async {
      final result = await f.call({
        'command': 'execution-output',
        'executionId': f.executionId,
      });
      expect(result['status'], 'ok');
      text = utf8.decode(
        (result['chunks'] as List)
            .expand<int>((chunk) => base64Decode(chunk['dataBase64'] as String))
            .toList(),
      );
      return text.contains('[REDACTED]');
    }, 'Controlled output did not reach the resident host');
    f.checkOutput(text);
  }

  for (final ending in ['revoke', 'expire', 'rotate', 'end']) {
    test(
      'encrypted mixed lifecycle: $ending keeps permissions, facts and native cleanup distinct',
      () async {
        final f = await fixture();
        await f.useThree();
        await f.until(
          () async =>
              f.service.owner.credentialExecutions.single.state == 'running',
          'Worker start was not durably recorded',
        );
        await assertRedactedOutput(f);
        final deadlines = {
          for (final r in f.service.owner.credentialUseRequests)
            r.id: r.credentialExpiresAt,
        };
        final originalPath = f.documentCopy!['path'] as String;
        final originalAttachment = f.documentCopy!['attachmentVersion'];
        switch (ending) {
          case 'revoke':
            f.service.owner.revoke(f.ordinaryRequest!);
            await f.service.owner.revokeCredentialUse(f.fillRequest!);
            await f.service.owner.revokeCredentialUse(f.commandRequest!);
          case 'expire':
            f.now = f.now.add(const Duration(minutes: 16));
          case 'rotate':
            await f.service.owner.rotateStringCredential(
              f.password.id,
              f.password.revision,
              'ROTATED_SYNTHETIC_PASSWORD',
            );
            await f.service.owner.rotateFileCredential(
              f.credentialFile.id,
              f.credentialFile.revision,
              fileName: 'rotated.ini',
              bytes: utf8.encode(
                '[default]\naws_access_key_id=ROTATED_ACCESS\naws_secret_access_key=ROTATED_SECRET\n',
              ),
            );
            f.item = await f.saveDocument('第二版虚构简历。', f.item);
          case 'end':
            final response = await f.hook({
              'hook_event_name': 'SessionEnd',
              'session_id': f.chat,
            });
            expect(response['systemMessage'], contains('ended'));
        }
        await f.service.owner.reconcileInformationFileDeliveries();
        if (ending == 'rotate') {
          expect(await File(originalPath).readAsString(), lifecycleDocument);
          final retry = f.delivery(await f.call(f.fetch('document-01')));
          expect(retry['path'], originalPath);
          expect(retry['attachmentVersion'], originalAttachment);
          final newer = f.delivery(await f.call(f.fetch('document-02')));
          expect(newer['attachmentVersion'], 2);
          expect(
            await File(newer['path'] as String).readAsString(),
            '第二版虚构简历。',
          );
          await assertRedactedOutput(f);
          f.service.owner.revoke(f.ordinaryRequest!);
          await f.service.agent.drainInformationFileDeliveries();
        } else {
          expect(await File(originalPath).exists(), false);
          final denied = await f.call(f.fetch('document-blocked'));
          expect(denied['errorCode'], switch (ending) {
            'end' => 'session_ended',
            'expire' => 'grant_expired',
            _ => 'not_authorized',
          });
        }
        final deniedFill = await f.call(f.fill('fill-blocked'));
        expect(deniedFill['errorCode'], switch (ending) {
          'expire' => 'grant_expired',
          'rotate' => 'credential_version_changed',
          _ => 'grant_revoked',
        });
        final deniedCommand = await f.call(f.run('command-blocked'));
        expect(deniedCommand['errorCode'], switch (ending) {
          'end' => 'session_required',
          'expire' => 'grant_expired',
          'rotate' => 'credential_version_changed',
          _ => 'grant_revoked',
        });
        final oldFill = await f.call(f.fill('fill-01'));
        expect(oldFill['fillState'], 'sent');
        expect(oldFill['duplicate'], true);
        expect(f.input.completed, hasLength(1));
        expect(
          f.service.owner.focusedFillOperations.single.state,
          FocusedFillState.sent,
        );
        expect(f.service.owner.credentialExecutions, hasLength(1));
        expect(await File('${f.root.path}/receiver-count').readAsString(), '1');
        expect(
          await f.material!.exists(),
          true,
          reason: 'An active Job still owns its credential file',
        );
        for (final r in f.service.owner.credentialUseRequests) {
          expect(
            r.state,
            'approved',
            reason: 'The recorded decision is separate from current permission',
          );
          expect(r.credentialExpiresAt, deadlines[r.id]);
          expect(f.service.owner.credentialGrantState(r), switch (ending) {
            'expire' => 'expired',
            'rotate' => 'version_invalidated',
            _ => 'revoked',
          });
        }
        await f.releaseReceiver();
        final execution = f.service.owner.credentialExecutions.single;
        expect(execution.state, 'exited');
        expect(execution.exitCode, 0);
        expect(execution.cleanupState, 'complete');
        expect(
          f.service.owner.informationFileDeliveries.every((r) => !r.holdsCopy),
          true,
        );
      },
      skip: !Platform.isWindows,
      timeout: const Timeout(Duration(minutes: 2)),
    );
  }

  test(
    'encrypted restart with an actual dead resident host preserves the active Job, facts and authority loss',
    () async {
      final f = await fixture();
      await f.useThree();
      await f.until(
        () async =>
            f.service.owner.credentialExecutions.single.state == 'running',
        'Worker start was not durably recorded',
      );
      final oldPath = f.documentCopy!['path'] as String;
      await f.restart();
      expect(await File(oldPath).exists(), false);
      expect(await f.material!.exists(), true);
      final interrupted = f.service.owner.credentialExecutions.single;
      expect(interrupted.state, 'unknown');
      expect(interrupted.exitCode, isNull);
      expect(interrupted.finishedAt, isNull);
      expect(interrupted.cleanupState, 'pending');
      expect(
        f.service.owner.focusedFillOperations.single.state,
        FocusedFillState.sent,
      );
      expect(
        f.service.owner.credentialUseRequests.every(
          (r) => f.service.owner.credentialGrantState(r) == 'revoked',
        ),
        true,
      );
      expect(
        (await f.call(f.fetch('restart-file')))['errorCode'],
        'not_authorized',
      );
      expect(
        (await f.call(f.fill('restart-fill')))['errorCode'],
        'session_required',
      );
      expect(
        (await f.call(f.run('restart-command')))['errorCode'],
        'session_required',
      );
      final retained = await f.hook({
        'hook_event_name': 'SessionStart',
        'source': 'resume',
        'session_id': f.chat,
      });
      expect(
        retained['hookSpecificOutput']['additionalContext'],
        contains('Retain'),
      );
      expect(
        (await f.call(f.fetch('resume-file')))['errorCode'],
        'not_authorized',
      );
      await f.releaseReceiver();
      final settled = f.service.owner.credentialExecutions.single;
      expect(settled.state, 'unknown');
      expect(settled.exitCode, isNull);
      expect(settled.finishedAt, isNull);
      expect(settled.cleanupState, 'complete');
      expect(await File('${f.root.path}/receiver-count').readAsString(), '1');
    },
    skip: !Platform.isWindows,
    timeout: const Timeout(Duration(minutes: 2)),
  );

  test(
    'a real encrypted writer lock prevents unaudited document, fill or command delivery and end still latches',
    () async {
      final f = await fixture();
      final deadlines = {
        for (final r in f.service.owner.credentialUseRequests)
          r.id: r.credentialExpiresAt,
      };
      await f.lockWrites();
      final file = await f.call(f.fetch('locked-file'));
      expect(file['errorCode'], 'audit_failed');
      final fill = await f.call(f.fill('locked-fill'));
      expect(fill['errorCode'], 'audit_unavailable');
      final command = await f.call(f.run('locked-command'));
      expect(command['errorCode'], 'audit_unavailable');
      expect(f.files.leases, isEmpty);
      expect(f.input.batches, 0);
      expect(f.service.owner.credentialExecutions, isEmpty);
      expect(await File('${f.root.path}/receiver-count').exists(), false);
      final end = await f.hook({
        'hook_event_name': 'SessionEnd',
        'session_id': f.chat,
      });
      expect(end['systemMessage'], contains('ended'));
      expect(
        (await f.call(f.fetch('ended-file')))['errorCode'],
        'session_ended',
      );
      expect(
        (await f.call(f.fill('ended-fill')))['errorCode'],
        'grant_revoked',
      );
      expect(
        (await f.call(f.run('ended-command')))['errorCode'],
        'session_required',
      );
      f.unlockWrites();
      await f.service.owner.retryAuditStorage();
      await f.service.owner.reconcileInformationFileDeliveries();
      expect(f.service.owner.auditStorageUnavailable, false);
      expect(f.service.owner.requests.single.state, AccessRequestState.revoked);
      expect(f.service.owner.credentialExecutions, isEmpty);
      expect(f.service.owner.focusedFillOperations, isEmpty);
      expect(f.service.owner.informationFileDeliveries.single.state, 'failed');
      for (final r in f.service.owner.credentialUseRequests) {
        expect(f.service.owner.credentialGrantState(r), 'revoked');
        expect(r.credentialExpiresAt, deadlines[r.id]);
      }
      expect(
        (await f.call(f.fetch('after-repair')))['errorCode'],
        'session_ended',
      );
      expect(f.files.leases, isEmpty);
    },
    skip: !Platform.isWindows,
    timeout: const Timeout(Duration(minutes: 2)),
  );
}
