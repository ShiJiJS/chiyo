from pathlib import Path
import subprocess,os,sys,json,hashlib
root=Path.cwd();out=root/'build/qa-issue30-terminal-core'/sys.argv[1];out.mkdir(parents=True,exist_ok=False)
sys.stdout.reconfigure(encoding='utf-8')
env=os.environ.copy();env.update(PUB_HOSTED_URL='https://pub.flutter-io.cn',FLUTTER_STORAGE_BASE_URL='https://storage.flutter-io.cn')
dart='C:/Users/Todoki/.local/share/flutter/bin/dart.bat'
paths=['packages/info_core','packages/info_storage','packages/info_cli','apps/desktop']
locks={root/p/'pubspec.lock':(root/p/'pubspec.lock').read_bytes() for p in paths}
norm=lambda b:b.replace(b'https://pub.flutter-io.cn',b'https://pub.dev').replace(b'\r\n',b'\n')
phases=[];sources=[]
def run(name,args,cwd):
    with (out/(name+'.txt')).open('wb') as f:p=subprocess.run(args,cwd=cwd,env=env,stdout=f,stderr=subprocess.STDOUT)
    phases.append(dict(phase=name,exitCode=p.returncode));print(json.dumps(phases[-1]),flush=True)
    if p.returncode:
        print((out/(name+'.txt')).read_text(encoding='utf-8',errors='replace')[-10000:],flush=True)
        raise RuntimeError(name)
try:
    for p,b in locks.items():p.write_bytes(b.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    cwd=root/'packages/info_core'
    run('pub',[dart,'pub','get','--enforce-lockfile','--offline'],cwd)
    changed=subprocess.check_output(['git','diff','--name-only'],cwd=root,text=True).splitlines()
    files=[p for p in changed if p.startswith('packages/info_core/') and p.endswith('.dart')]+['packages/info_core/lib/credential_execution_audit.dart','packages/info_core/test/credential_execution_audit_test.dart']
    run('format',[dart,'format',*[str(root/p) for p in files]],cwd)
    for p in files:
        b=(root/p).read_bytes();(out/(Path(p).name+'.txt')).write_bytes(b);sources.append(dict(path=p,sha256=hashlib.sha256(b).hexdigest()))
    run('analyze',[dart,'analyze','--fatal-infos'],cwd)
    run('test',[dart,'test','-r','expanded','test/credential_execution_audit_test.dart','test/credential_operations_test.dart','test/credential_use_test.dart','test/credential_history_retention_test.dart','test/focused_fill_lifecycle_test.dart'],cwd)
finally:
    unexpected=[]
    for p,b in locks.items():
        if norm(p.read_bytes())==norm(b):p.write_bytes(b)
        else:unexpected.append(str(p))
    (out/'result.json').write_text(json.dumps(dict(phases=phases,sources=sources,unexpectedLocks=unexpected),indent=2)+'\n',encoding='utf-8')
    assert not unexpected,unexpected
