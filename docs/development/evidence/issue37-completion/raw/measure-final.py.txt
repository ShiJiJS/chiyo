from pathlib import Path
import subprocess,sys,os,json,hashlib
sys.stdout.reconfigure(encoding='utf-8')
root=Path.cwd();package=root/'packages/info_storage';out=root/'build/qa-issue37-capacity-outcomes'/sys.argv[1];out.mkdir(parents=True,exist_ok=False)
dart='C:/Users/Todoki/.local/share/flutter/bin/dart.bat';env=os.environ.copy();env['PUB_HOSTED_URL']='https://pub.flutter-io.cn';env['FLUTTER_STORAGE_BASE_URL']='https://storage.flutter-io.cn'
locks={root/p/'pubspec.lock':(root/p/'pubspec.lock').read_bytes() for p in ['packages/info_core','packages/info_storage','packages/info_cli','apps/desktop']}
phases=[];runs=[];artifacts=[]
def run(name,args,cwd=package):
    with (out/(name+'.txt')).open('wb') as log:r=subprocess.run(args,cwd=cwd,env=env,stdout=log,stderr=subprocess.STDOUT)
    phases.append({'phase':name,'exitCode':r.returncode});print(json.dumps(phases[-1]),flush=True)
    if r.returncode:
        print((out/(name+'.txt')).read_text(encoding='utf-8',errors='replace')[-8000:],flush=True);raise RuntimeError(name)
try:
    for p,b in locks.items():p.write_bytes(b.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    run('pub',[dart,'pub','get','--enforce-lockfile','--offline'])
    run('analyze',[dart,'analyze','--fatal-infos'])
    run('build',[dart,'build','cli','--target=tool/measure_information_library.dart','--output='+str(out/'aot')])
    bundle=out/'aot/bundle';executables=list((bundle/'bin').glob('*.exe'));assert len(executables)==1,executables
    exe=executables[0]
    for p in sorted(bundle.rglob('*')):
        if p.is_file():artifacts.append({'path':p.relative_to(bundle).as_posix(),'bytes':p.stat().st_size,'sha256':hashlib.sha256(p.read_bytes()).hexdigest()})
    for mode,scenario in [('aot','near-limit'),('aot','history-limit')]:
        name=mode+'-'+scenario
        args=[dart,'run','tool/measure_information_library.dart',scenario] if mode=='jit' else [str(exe),scenario]
        run(name,args,package if mode=='jit' else exe.parent)
        lines=(out/(name+'.txt')).read_text(encoding='utf-8').splitlines();value=json.loads(lines[-1])
        assert value['temporaryOutputRemoved'] is True and value['result']['allVersionsExact'] is True and value['result']['restoredFactsOnly'] is True
        value['run']=name;runs.append(value)
        r=value['result'];print(json.dumps({'run':name,'maxRssMiB':round(r['maxRss']/1048576,1),'metadataBytes':r['metadataBytes'],'backupFileBytes':r['backupFileBytes'],'totalMs':r['totalMs']}),flush=True)
finally:
    unexpected=[]
    norm=lambda b:b.replace(b'https://pub.flutter-io.cn',b'https://pub.dev').replace(b'\r\n',b'\n')
    for p,b in locks.items():
        if norm(p.read_bytes())!=norm(b):unexpected.append(str(p))
        else:p.write_bytes(b)
    (out/'result.json').write_text(json.dumps({'phases':phases,'runs':runs,'artifacts':artifacts,'unexpectedLocks':unexpected},ensure_ascii=False,indent=2)+'\n',encoding='utf-8',newline='\n')
    assert not unexpected,unexpected
