from pathlib import Path
import hashlib, json, shutil, subprocess

root = Path.cwd()
qa = root / 'build/qa-issue37-capacity-outcomes'
source = subprocess.check_output(['git', 'rev-parse', 'HEAD'], text=True).strip()
assert source == '8d6349fe526ccb8d9d2446ba5cd42c219fae17c7'
assert json.loads((qa / 'full-01/result.json').read_text())['exitCode'] == 0
dest = root / 'docs/development/evidence/issue37-completion'
(dest / 'raw').mkdir(parents=True, exist_ok=False)
raw = []
def copy(p, relative):
    target = dest / 'raw' / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(p, target)
    b = target.read_bytes()
    raw.append({'path': target.relative_to(dest).as_posix(), 'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()})

for phase in [*(f'round-{i:02}' for i in range(1, 11)), 'full-01', 'measurement-01', 'measurement-02', 'legacy-01', 'legacy-02']:
    for p in sorted((qa / phase).iterdir()):
        if p.is_file() and p.suffix in {'.txt', '.json'}:
            copy(p, Path(phase) / p.name)
for p in sorted(qa.glob('*.py')):
    copy(p, Path(p.name + '.txt'))

paths = set()
for revision in ['5af4e4d4eac30d440bc097b163bc8f82dea6ed3a', source]:
    paths.update(subprocess.check_output(['git', 'diff-tree', '--no-commit-id', '--name-only', '-r', revision], text=True).splitlines())
paths.update(p + '/pubspec.lock' for p in ['packages/info_core', 'packages/info_storage', 'packages/info_cli', 'apps/desktop'])
sources = []
for p in sorted(paths):
    b = subprocess.check_output(['git', 'show', source + ':' + p])
    sources.append({'path': p, 'bytes': len(b), 'sha256': hashlib.sha256(b).hexdigest()})
record = {

    'earlierMeasurementRevision': '5af4e4d4eac30d440bc097b163bc8f82dea6ed3a',

    'raw': raw, 'sourceGitBytes': sources,
}
(dest / 'verification.json').write_text(json.dumps(record, ensure_ascii=False, indent=2) + '\n', encoding='utf-8', newline='\n')

p.write_text(p.read_text(encoding='utf-8') + '\ndocs/development/evidence/issue37-completion/raw/** -text -diff\n', encoding='utf-8', newline='\n')
