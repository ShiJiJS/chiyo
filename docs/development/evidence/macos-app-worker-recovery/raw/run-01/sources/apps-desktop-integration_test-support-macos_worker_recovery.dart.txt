import 'dart:async';
import 'dart:convert';
import 'dart:io';
import 'package:info_cli/agent_client.dart';
import 'package:info_core/info_core.dart';
import 'package:info_desktop/desktop_bootstrap.dart';

void _check(bool value, String message) {
  if (!value) throw StateError(message);
}

Future<void> _until(FutureOr<bool> Function() done, String message) async {
  final end = DateTime.now().add(const Duration(seconds: 75));
  while (!await done()) {
    _check(DateTime.now().isBefore(end), message);
    await Future<void>.delayed(const Duration(milliseconds: 40));
  }
}

/// Actual sandboxed Flutter host and bundled worker, actual Apple SSH. Only
/// approvals, synthetic keys, event triggers and loopback remote command vary.
Future<void> runMacWorkerRecovery(
  DesktopBootstrap bootstrap,
  String directory,
  String mode,
  File report,
) async {
  final owner = bootstrap.library!.service.owner;
  final privateFile = File('$directory/worker-association.json');
  if (mode == 'worker-recover') {
    final old = jsonDecode(await privateFile.readAsString()) as Map;
    var execution = owner.credentialExecutions.single;
    _check(
      execution.state == 'unknown' &&
          execution.exitCode == null &&
          execution.finishedAt == null &&
          execution.cleanupState == 'needs_check',
      'Restart fabricated execution completion',
    );
    final materials = Directory(execution.materialLease!.path);
    _check(
      await materials.exists(),
      'Active credential materials were removed',
    );
    _check(
      CredentialProcessJob.isActive(
            execution.jobName!,
            workerPid: execution.workerPid,
            workerCreatedAt: execution.workerCreatedAt,
          ) ==
          true,
      'Target group did not survive application death',
    );
    _check(
      !await File(old['copyPath'] as String).exists() &&
          owner.informationFileDeliveries.every((d) => !d.holdsCopy),
      'Old ordinary attachment copy was not reclaimed independently',
    );
    final client = AgentClient(
      bootstrap.boundPort!,
      sessionSecret: old['session'] as String,
      hostedCredentialExecutions: true,
    );
    try {
      final run = await client.execute(
        Map<String, dynamic>.from(old['run'] as Map),
      );
      final fetch = await client.execute(
        Map<String, dynamic>.from(old['fetch'] as Map),
      );
      _check(
        run['status'] != 'accepted' && fetch['status'] != 'ok',
        'Old association revived a command or attachment',
      );
      _check(owner.credentialExecutions.length == 1, 'Command replayed');
      await report.writeAsString(
        jsonEncode({
          'mode': mode,
          'phase': 'active-material-retained',
          'targetGroupAlive': true,
          'ordinaryCopyRemoved': true,
          'credentialMaterialsRetained': true,
          'executionState': execution.state,
          'oldRunStatus': run['status'],
          'oldRunError': run['errorCode'],
          'oldFetchStatus': fetch['status'],
          'oldFetchError': fetch['errorCode'],
        }),
        flush: true,
      );
      // The Python driver only releases the SSH server after inspecting this
      // checkpoint. Reconciliation must wait for actual target-group completion.
      await _until(() async {
        await owner.reconcileCredentialMaterialCleanup();
        return owner.credentialExecutions.single.cleanupState == 'complete';
      }, 'Materials did not settle after receiver completion');
      execution = owner.credentialExecutions.single;
      _check(
        execution.state == 'unknown' &&
            execution.exitCode == null &&
            execution.finishedAt == null &&
            !await materials.exists() &&
            CredentialProcessJob.isActive(
                  execution.jobName!,
                  workerPid: execution.workerPid,
                  workerCreatedAt: execution.workerCreatedAt,
                ) ==
                false,
        'Cleanup changed facts or left active materials',
      );
      await report.writeAsString(
        jsonEncode({
          'mode': mode,
          'phase': 'complete',
          'ordinaryCopyRemoved': true,
          'activeMaterialsWereRetained': true,
          'materialsRemovedAfterGroupExit': true,
          'executionState': execution.state,
          'exitCode': execution.exitCode,
          'finishedAt': execution.finishedAt,
          'executionCount': owner.credentialExecutions.length,
          'oldAuthorityRefused': true,
        }),
        flush: true,
      );
    } finally {
      client.close();
    }
    await bootstrap.runtime!.quit();
    return;
  }

  final port = Platform.environment['INFO_MAC_SSH_PORT']!;
  final key = Platform.environment['INFO_MAC_SSH_KEY']!;
  final host = Platform.environment['INFO_MAC_SSH_HOST']!;
  await owner.configureSshKnownHosts(
    'Synthetic loopback only',
    utf8.encode('[127.0.0.1]:$port $host\n'),
  );
  var credential = await owner.createFileCredential(
    name: 'Synthetic lifecycle SSH',
    publicPurpose: '',
    fileName: 'synthetic.key',
    bytes: base64Decode(key),
  );
  credential = await owner.updateCredentialMetadata(
    credential.id,
    credential.revision,
    name: credential.name,
    publicPurpose: '',
    privateNote: '',
    suggestedBinding: 'ssh_key',
    discoverable: true,
  );
  final source = File('$directory/synthetic-attachment.bin');
  await source.writeAsBytes([0, 255, 17, 10]);
  final draft = await owner.beginInformationFileDraft();
  final staged = await owner.stageInformationFile(draft, source.path);
  final item = await owner.commitInformationDraft(
    draft,
    'synthetic-worker-file',
    InformationDraftInput(
      name: 'Synthetic worker file',
      description: '',
      fields: [InformationDraftField.stagedFile('File', '', staged.id)],
    ),
  );
  await owner.releaseInformationFileDraft(draft);
  final client = AgentClient(
    bootstrap.boundPort!,
    hostedCredentialExecutions: true,
  );
  await client.execute({'command': 'session', 'chatLabel': 'Synthetic worker'});
  final launch = {
    'receiver': 'ssh_identity',
    'executable': '/usr/bin/ssh',
    'argv': [
      '-T',
      '-o',
      'BatchMode=yes',
      '-i',
      {'file': 'ssh_key'},
      '-p',
      port,
      'fixture@127.0.0.1',
      '/fixture/wait',
    ],
    'cwd': directory,
    'env': <String, Object?>{},
  };
  final ref = FieldRef(item.id, item.fields.single.id);
  final request = await client.execute({
    'command': 'request-use',
    'protocolVersion': 2,
    'purpose': 'Synthetic worker restart',
    'credentials': [
      {'alias': 'ssh_key', 'credentialId': credential.id, 'version': 1},
    ],
    'launch': launch,
    'readFields': [ref.toJson()],
  });
  _check(request['status'] == 'pending', 'Request did not reach owner');
  final id = request['requestId'] as String;
  await owner.decideCredentialUse(id, approveCredential: true, fields: [ref]);
  final run = {
    'command': 'run-use',
    'protocolVersion': 2,
    'requestId': id,
    'operationId': 'one-command-only',
    'launch': launch,
  };
  final accepted = await client.execute(run);
  _check(accepted['status'] == 'accepted', 'Hosted execution was not accepted');
  await _until(
    () => owner.credentialExecutions.single.state == 'running',
    'Worker never reached durable running state',
  );
  final fetch = {
    'command': 'file-fetch',
    'requestId': id,
    'itemId': item.id,
    'fieldId': item.fields.single.id,
    'operationId': 'one-file',
  };
  final copy = await client.execute(fetch);
  _check(copy['status'] == 'ok', 'Ordinary file delivery failed');
  final copyPath = (copy['delivery'] as Map)['path'] as String;
  _check(
    base64Encode(await File(copyPath).readAsBytes()) ==
        base64Encode([0, 255, 17, 10]),
    'Delivered bytes changed',
  );
  var execution = owner.credentialExecutions.single;
  _check(
    execution.supervisorPid == pid &&
        CredentialProcessIdentity.isActive(
              execution.workerPid!,
              execution.workerCreatedAt!,
            ) ==
            true &&
        await Directory(execution.materialLease!.path).exists(),
    'Execution is not owned by this Flutter host',
  );
  await privateFile.writeAsString(
    jsonEncode({
      'session': client.sessionSecret,
      'run': run,
      'fetch': fetch,
      'copyPath': copyPath,
    }),
    flush: true,
  );
  _check(
    (await Process.run('/bin/chmod', ['600', privateFile.path])).exitCode == 0,
    'Private fixture permissions unavailable',
  );
  final crashWorker = Platform.environment['INFO_MAC_CRASH_WORKER'] == 'true';
  if (crashWorker) {
    _check(
      Process.killPid(execution.workerPid!, ProcessSignal.sigkill),
      'Owned worker crash injection failed',
    );
    await _until(
      () =>
          CredentialProcessIdentity.isActive(
            execution.workerPid!,
            execution.workerCreatedAt!,
          ) ==
          false,
      'Worker did not exit',
    );
  }
  execution = owner.credentialExecutions.single;
  await report.writeAsString(
    jsonEncode({
      'mode': mode,
      'phase': 'ready',
      'port': bootstrap.boundPort,
      'workerCrashInjected': crashWorker,
      'supervisorIsFlutter': execution.supervisorPid == pid,
      'workerPid': execution.workerPid,
      'workerCreatedAt': execution.workerCreatedAt,
      'executionState': execution.state,
      'materialsPresent': await Directory(
        execution.materialLease!.path,
      ).exists(),
    }),
    flush: true,
  );
  await _until(
    () async => await File('$directory/never-finish').exists(),
    'Driver did not inject application crash',
  );
}
