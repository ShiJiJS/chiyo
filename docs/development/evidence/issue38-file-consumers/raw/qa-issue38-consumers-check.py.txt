from pathlib import Path
import os,sys,json,subprocess
sys.stdout.reconfigure(encoding='utf-8')

    for p,b in locks.items():p.write_bytes(b.replace(b'https://pub.dev',b'https://pub.flutter-io.cn'))
    run('packages/info_core','core-pub',['pub','get','--enforce-lockfile','--offline'])
    run('packages/info_core','core-format',['format','lib/credential_runtime_access.dart'])

    run('apps/desktop','desktop-analyze',['analyze','--fatal-infos'])
    run('apps/desktop','desktop-tests',['test','--no-pub',*(['test/'] if 'all' in sys.argv[2:] else ['test/information_file_consumers_test.dart','test/recycle_test.dart','test/revisions_test.dart','test/proposal_workspace_test.dart','test/update_proposals_test.dart','test/credential_use_widget_test.dart','test/credential_reuse_widget_test.dart','test/credential_migration_widget_test.dart']),'--reporter','expanded'])
    if 'render' in sys.argv[2:]:run('apps/desktop','render',['test','--no-pub','--dart-define=ATTACHMENT_CONSUMERS_RENDER_DIRECTORY='+str(out/'screens'),'tool/render_information_file_consumers_test.dart','--reporter','expanded'])
