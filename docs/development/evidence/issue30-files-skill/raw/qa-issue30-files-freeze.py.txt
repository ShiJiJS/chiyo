from pathlib import Path
import hashlib, json, shutil, subprocess, sys
sys.stdout.reconfigure(encoding='utf-8')
root = Path.cwd()
assert not subprocess.check_output(['git', 'status', '--porcelain'], cwd=root).strip()
source = subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=root, text=True).strip()
dest = root / 'docs/development/evidence/issue30-files-skill'
dest.mkdir(parents=True, exist_ok=False)
raw = dest / 'raw'
raw.mkdir()
qa = root / 'build/qa-issue30-files'
for run in ['adapter-01','adapter-02','adapter-03','adapter-04','adapter-05','adapter-06','adapter-07','adapter-08','regression-01']:
    target = raw / run
    target.mkdir()
    for path in sorted((qa / run).iterdir()):
        if path.is_file() and path.suffix in ['.json', '.txt']:
            shutil.copyfile(path, target / path.name)
for name in ['qa-issue30-files.py','qa-issue30-files-regression.py','qa-issue30-files-freeze.py']:
    shutil.copyfile(root / 'build' / name, raw / (name + '.txt'))
files = [dict(path=p.relative_to(dest).as_posix(), byteLength=p.stat().st_size, sha256=hashlib.sha256(p.read_bytes()).hexdigest()) for p in sorted(raw.rglob('*')) if p.is_file()]
sources = subprocess.check_output(['git','diff','--name-only','cf75512',source], cwd=root,text=True).splitlines()
sources += [p+'/pubspec.lock' for p in ['packages/info_cli','packages/info_core','packages/info_storage','apps/desktop']]
sourceFacts=[]
for relative in sources:
    committed = subprocess.check_output(['git','show',f'{source}:{relative}'],cwd=root)
    workspace = (root/relative).read_bytes()
    assert committed.replace(b'\r\n',b'\n') == workspace.replace(b'\r\n',b'\n'), relative
    sourceFacts.append(dict(path=relative,gitSha256=hashlib.sha256(committed).hexdigest(),workingTreeSha256=hashlib.sha256(workspace).hexdigest()))
binaries=[]
for path in sorted((qa/'adapter-08/codex-preview').glob('*.exe')):
    binaries.append(dict(path=path.relative_to(root).as_posix(),byteLength=path.stat().st_size,sha256=hashlib.sha256(path.read_bytes()).hexdigest()))
manifest=dict(sourceCommit=source,scope='Windows independent adapters and native file/process contracts; Memory repository and fake focused-input driver; not real Codex events or final ZIP',counts=dict(aotSelectedPassed=19,aotSelectedSkipped=0,sourceCliPassed=92,sourceCliSkipped=13),sources=sourceFacts,binaries=binaries,artifacts=files)
(dest/'verification.json').write_text(json.dumps(manifest,ensure_ascii=False,indent=2)+'\n',encoding='utf-8')
print(json.dumps(dict(sourceCommit=source,artifacts=len(files),sources=len(sourceFacts),binaries=len(binaries)),indent=2))
