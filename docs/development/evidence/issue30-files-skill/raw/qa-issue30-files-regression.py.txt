from pathlib import Path
import os, sys, json, subprocess
sys.stdout.reconfigure(encoding='utf-8')
root = Path.cwd()
out = root / 'build/qa-issue30-files' / sys.argv[1]
out.mkdir(parents=True, exist_ok=False)
dart = 'C:/Users/Todoki/.local/share/flutter/bin/dart.bat'
env = os.environ.copy()
env.update(PUB_HOSTED_URL='https://pub.flutter-io.cn', FLUTTER_STORAGE_BASE_URL='https://storage.flutter-io.cn')
env.pop('INFO_TEST_BUNDLE', None)
env.pop('INFO_TEST_DELIVERY', None)
locks = {root / p / 'pubspec.lock': (root / p / 'pubspec.lock').read_bytes() for p in ['packages/info_core', 'packages/info_storage', 'packages/info_cli', 'apps/desktop']}
phases = []
norm = lambda b: b.replace(b'https://pub.flutter-io.cn', b'https://pub.dev').replace(b'\r\n', b'\n')
def run(name, cmd, cwd):
    with (out / (name + '.txt')).open('wb') as log:
        result = subprocess.run(cmd, cwd=cwd, env=env, stdout=log, stderr=subprocess.STDOUT)
    phases.append(dict(phase=name, exitCode=result.returncode))
    print(json.dumps(phases[-1]), flush=True)
    if result.returncode:
        print((out / (name + '.txt')).read_text(encoding='utf-8', errors='replace')[-15000:], flush=True)
        raise RuntimeError(name)
try:
    for p, b in locks.items(): p.write_bytes(b.replace(b'https://pub.dev', b'https://pub.flutter-io.cn'))
    cli = root / 'packages/info_cli'
    run('pub', [dart, 'pub', 'get', '--enforce-lockfile', '--offline'], cli)
    run('analyze', [dart, 'analyze', '--fatal-infos'], cli)
    run('full-cli-source', [dart, 'test', '-r', 'expanded'], cli)
    run('powershell-parse', ['powershell.exe', '-NoProfile', '-NonInteractive', '-Command', "$ErrorActionPreference='Stop'; foreach ($name in @('scripts/package-windows.ps1','scripts/test-windows-package.ps1')) { $taskTokens=$null; $taskErrors=$null; $null=[System.Management.Automation.Language.Parser]::ParseFile((Join-Path (Get-Location) $name),[ref]$taskTokens,[ref]$taskErrors); if ($taskErrors.Count) { throw ($taskErrors | Out-String) }; Write-Output $name }"], root)
finally:
    unexpected = []
    for p, b in locks.items():
        if norm(p.read_bytes()) != norm(b): unexpected.append(str(p))
        else: p.write_bytes(b)
    (out / 'result.json').write_text(json.dumps(dict(phases=phases, unexpectedLocks=unexpected), indent=2) + '\n', encoding='utf-8')
    assert not unexpected, unexpected
